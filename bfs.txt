#include <iostream>
#include <vector>
#include <queue>
#include <omp.h>

using namespace std;

// Number of nodes in the graph
const int N = 6;

vector<vector<int>> graph(N); // Adjacency list
vector<bool> visited(N, false);

void parallelBFS(int start) {
    queue<int> q;
    q.push(start);
    visited[start] = true;

    while (!q.empty()) {
        int size = q.size();

        // Create a temporary queue to store next level nodes
        vector<int> tempQueue;

        // Parallel processing of current level nodes
        #pragma omp parallel for shared(q, graph, visited, tempQueue)
        for (int i = 0; i < size; ++i) {
            int node;

            // To avoid race condition while accessing queue
            #pragma omp critical
            {
                node = q.front();
                q.pop();
                cout << "Visited Node: " << node << " by thread " << omp_get_thread_num() << endl;
            }

            // Traverse all neighbors
            for (int neighbor : graph[node]) {
                if (!visited[neighbor]) {
                    #pragma omp critical
                    {
                        if (!visited[neighbor]) {
                            visited[neighbor] = true;
                            tempQueue.push_back(neighbor);
                        }
                    }
                }
            }
        }

        // Push next level nodes into the main queue
        for (int node : tempQueue) {
            q.push(node);
        }
    }
}

int main() {
    // Creating a simple undirected graph
    graph[0] = {1, 2};
    graph[1] = {0, 3, 4};
    graph[2] = {0, 4};
    graph[3] = {1, 5};
    graph[4] = {1, 2, 5};
    graph[5] = {3, 4};

    cout << "Parallel BFS traversal starting from node 0:\n";

    parallelBFS(0);

    return 0;
}
