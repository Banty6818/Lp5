#include <iostream>
#include <omp.h>
#include <cstdlib>

void parallelBubbleSort(int *arr, int n)
{
  for (int i = 0; i < n; i++)
  {
    int start = i % 2; // Even or odd phase
#pragma omp parallel for
    for (int j = start; j < n - 1; j += 2)
    {
      int tid = omp_get_thread_num(); // Get thread ID
      std::cout << "Thread " << tid << " comparing index " << j << " and " << j + 1 << std::endl;

      if (arr[j] > arr[j + 1])
      {
        // Swap
        int temp = arr[j];
        arr[j] = arr[j + 1];
        arr[j + 1] = temp;
        std::cout << "Thread " << tid << " swapped " << arr[j + 1] << " and " << arr[j] << std::endl;
      }
    }
  }
}

void printArray(int *arr, int n)
{
  for (int i = 0; i < n; i++)
  {
    std::cout << arr[i] << " ";
  }
  std::cout << std::endl;
}

int main()
{
  int n;
  std::cout << "Enter number of elements: ";
  std::cin >> n;

  int *arr = new int[n];

  std::cout << "Enter " << n << " elements:\n";
  for (int i = 0; i < n; i++)
  {
    std::cin >> arr[i];
  }

  std::cout << "\nStarting Parallel Bubble Sort with thread output...\n\n";
  parallelBubbleSort(arr, n);

  std::cout << "\nSorted Array:\n";
  printArray(arr, n);

  delete[] arr;
  return 0;
}
