#include <iostream>
#include <omp.h>
#include <limits>

int main()
{
  int n;

  std::cout << "Enter the number of elements: ";
  std::cin >> n;

  int *arr = new int[n];

  std::cout << "Enter " << n << " elements:\n";
  for (int i = 0; i < n; i++)
    std::cin >> arr[i];

  int min_val = INT_MAX;
  int max_val = INT_MIN;
  long long sum = 0;

#pragma omp parallel for reduction(min : min_val) reduction(max : max_val) reduction(+ : sum)
  for (int i = 0; i < n; i++)
  {
    if (arr[i] < min_val)
      min_val = arr[i];
    if (arr[i] > max_val)
      max_val = arr[i];
    sum += arr[i];
  }

  double average = static_cast<double>(sum) / n;

  std::cout << "\nResults using Parallel Reduction:\n";
  std::cout << "Minimum: " << min_val << "\n";
  std::cout << "Maximum: " << max_val << "\n";
  std::cout << "Sum: " << sum << "\n";
  std::cout << "Average: " << average << "\n";

  delete[] arr;
  return 0;
}
